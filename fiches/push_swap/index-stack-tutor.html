int index_stack(t_stack *a)
{
int n;
int *vals;
t_stack *cur;
n = stack_size(a);
vals = malloc(sizeof(int) * n);
if (!vals) return (0);
copy_vals(a, vals);
cur = a;
while (cur)
{
cur->val = compute_rank(cur->val, vals, n);
cur = cur->next;
}
free(vals);
return (1);
}
