index_stack()
int index_stack(t_stack *a)
{
int n;
int *vals;
t_stack *cur;
n = stack_size(a);
vals = malloc(sizeof(int)*n);
copy_vals(a, vals);
cur = a;
while (cur)
{
cur->val = compute_rank(cur->val, vals, n);
cur = cur->next;
}
free(vals);
return (1);
}
helpers (static)
// copie pile → tableau
static void copy_vals(t_stack *a, int *vals)
{
int i = 0;
while (a)
{
vals[i] = a->val;
a = a->next;
i++;
}
}
// rang = # vals < target
static int compute_rank(int target, int *vals, int n)
{
int rank = 0;
int j = 0;
while (j < n)
{
if (vals[j] < target)
rank++;
j++;
}
return (rank);
}
