01
fprime
Décomposer un entier en facteurs premiers, séparés par *
✓ Rendu
fprime.c 
#include <stdio.h>
#include <stdlib.h>

int main(int argc, char **argv)
{
	if (argc != 2)
	{
		printf("\n");
		return 0;
	}
	int n = atoi(argv[1]);
	if (n == 1)
	{
		printf("1\n");
		return 0;
	}
	int div = 2;
	while (n > 1)
	{
		if (n % div == 0)
		{
			printf("%d", div);
			n /= div;
			if (n > 1)
				printf("*");
		}
		else
			div++;
	}
	printf("\n");
	return 0;
}
Mantra
div=2, tant que n>1 : divise ou incrémente
Pattern
Trial division sans remonter : quand div divise plus → on incrémente. Les composés ne peuvent pas passer (leurs facteurs premiers ont déjà été consommés).
Piège : oublier le cas n == 1 → affiche rien (alors qu'il faut "1\n").

ft_itoa
Convertir un int en string (avec malloc)
✓ Rendu
ft_itoa.c 
#include <stdlib.h>

char *ft_itoa(int nbr)
{
	char	*result;
	long	n;
	long	tmp;
	int		len = 1;
	int		signe = 1;
	int		i;

	n = nbr;
	if (nbr == 0)
	{
		result = malloc(2);
		if (!result) return (NULL);
		result[0] = '0'; result[1] = '\0';
		return (result);
	}
	if (n < 0) { signe = -1; n = -n; len += 1; }
	tmp = n;
	while (tmp >= 10) { len++; tmp /= 10; }
	result = malloc(len + 1);
	if (!result) return (NULL);
	result[len] = '\0';
	i = len - 1;
	while (n > 0)
	{
		result[i--] = n % 10 + '0';
		n /= 10;
	}
	if (signe == -1) result[0] = '-';
	return (result);
}
Mantra
n=long, 0 à part, signe, compte, malloc, remplis de droite à gauche
Pattern
n = long impératif : sinon -INT_MIN overflow. Edge cases explicites : 0, INT_MIN. Remplissage bas vers haut (modulo 10).
Piège : n = -n sur INT_MIN avec int → UB. D'où le long n.
→ Fiche révision ↯ Step debugger

ft_list_foreach
Appliquer f(data) à chaque nœud de la liste
✓ Rendu
ft_list_foreach.c 
#include "ft_list.h"

void	ft_list_foreach(t_list *begin_list, void (*f)(void *))
{
	while (begin_list)
	{
		f(begin_list->data);
		begin_list = begin_list->next;
	}
}
Mantra
while (node) { f(node->data); node = node->next; }
Pattern
Le plus simple des 7. Itération linéaire avec callback. while (begin_list) traite aussi le cas liste vide (entre jamais dans le while).
Piège : oublier begin_list = begin_list->next → boucle infinie sur le premier nœud.

04
rev_wstr
Afficher les mots d'une string dans l'ordre inverse
✓ Rendu
rev_wstr.c 
#include <unistd.h>

int main(int argc, char **argv)
{
	char	*str = argv[1];
	int		i = 0;
	int		end = 0;

	if (argc != 2)
	{
		write(1, "\n", 1);
		return (0);
	}
	while (str[i]) i++;
	i--;
	while (i >= 0)
	{
		end = i;
		while (i >= 0 && str[i] != ' ' && str[i] != '\t')
			i--;
		write(1, &str[i + 1], end - i);
		if (i >= 0)
			write(1, " ", 1);
		i--;
	}
	write(1, "\n", 1);
	return (0);
}
Mantra
Droite→gauche : marque end, recule sur le mot, write, saute l'espace
Pattern
Deux pointeurs qui convergent. Pas de malloc — on write directement les substrings via offset &str[i+1] + longueur end-i.
Piège : oublier i-- après l'espace sauté → re-entre dans le même mot → boucle infinie.
→ Fiche révision ↯ Step debugger
05
rostring
Rotation à gauche d'un mot (1er mot → fin), espaces normalisés à 1
✓ Rendu
rostring.c 
#include <unistd.h>

int main(int argc, char **argv)
{
	int		i = 0;
	char	*str;
	int		start, end, w_start;
	int		printed = 0;

	if (argc < 2) { write(1, "\n", 1); return (0); }
	str = argv[1];
	while (str[i] == ' ' || str[i] == '\t') i++;
	start = i;
	while (str[i] && str[i] != ' ' && str[i] != '\t') i++;
	end = i;
	while (str[i])
	{
		while (str[i] == ' ' || str[i] == '\t') i++;
		if (str[i])
		{
			if (printed) write(1, " ", 1);
			w_start = i;
			while (str[i] && str[i] != ' ' && str[i] != '\t') i++;
			write(1, &str[w_start], i - w_start);
			printed = 1;
		}
	}
	if (printed) write(1, " ", 1);
	write(1, &str[start], end - start);
	write(1, "\n", 1);
	return (0);
}
Mantra
Mémorise le 1er mot (start/end), print les autres, puis le 1er à la fin
Pattern
Flag printed pour gérer les espaces sans doublon. Le 1er mot n'est jamais writeé tout de suite — on mémorise juste ses bornes et on le sort en dernier.
Piège : printer un espace avant CHAQUE mot → espace leading parasite. D'où le flag printed.
→ Fiche révision ↯ Step debugger

sort_int_tab
Bubble sort in-place sur tableau d'entiers
✓ Rendu
sort_int_tab.c 
void	sort_int_tab(int *tab, unsigned int size)
{
	unsigned int	i;
	int				tmp;
	int				swaped;

	if (size < 2)
		return ;
	swaped = 1;
	while (swaped)
	{
		swaped = 0;
		i = 0;
		while (i < size - 1)
		{
			if (tab[i] > tab[i + 1])
			{
				tmp = tab[i];
				tab[i] = tab[i + 1];
				tab[i + 1] = tmp;
				swaped = 1;
			}
			i++;
		}
	}
}
Mantra
Guard size<2, flag en haut de passe, swap voisins jusqu'à stabilité
Pattern
swaped = 0; i = 0; en HAUT de chaque passe (groupés). Flag = "qqch a bougé CETTE passe". Sort dès qu'une passe ne touche à rien.
Piège : size est unsigned → si size == 0, size - 1 underflow à UINT_MAX. D'où le guard.
→ Fiche révision ↯ Step debugger
07
sort_list
Bubble sort sur liste chaînée avec cmp custom
✓ Rendu
sort_list.c 
#include "list.h"

t_list *sort_list(t_list *lst, int (*cmp)(int, int))
{
	if (!lst || !lst->next)
		return (lst);
	int swapped = 1;
	t_list *cur;
	int tmp;

	while (swapped)
	{
		swapped = 0;
		cur = lst;
		while (cur->next)
		{
			if (!cmp(cur->data, cur->next->data))
			{
				tmp = cur->data;
				cur->data = cur->next->data;
				cur->next->data = tmp;
				swapped = 1;
			}
			cur = cur->next;
		}
	}
	return (lst);
}
Mantra
!cmp(a, b) = swap des data (pas des nœuds). La tête ne bouge jamais.
Pattern
Même algo que sort_int_tab mais sur liste : i++ devient cur = cur->next, size - 1 devient cur->next != NULL. Swap des ->data, les nœuds restent en place.
Piège : oublier le ! devant cmp(...) → cmp != 0 signifie "bien ordonné", donc sans ! on swap quand c'est déjà trié. Tri inversé.
→ Fiche révision ↯ Step debugger
