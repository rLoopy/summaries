max
Trouver le plus grand int dans un tableau
✓ Completed
max.c 
int max(int* tab, unsigned int len)
{
	unsigned int i = 0;
	if (len == 0)
		return 0;
	int max = tab[i];
	while (i < len)
	{
		if (tab[i] > max)
			max = tab[i];
		i++;
	}
	return (max);
}
Mantra
Init max = tab[0], scan len fois, compare & update
Pattern
Parcours tableau avec accumulateur. Vérifier len == 0 AVANT d'accéder à tab[0].
Piège : init max = 0 au lieu de tab[0] → faux avec tableau tout négatif.

ft_strcmp
Comparer deux strings
✓ Completed
ft_strcmp.c 
int	ft_strcmp(char *s1, char *s2)
{
	int i = 0;
	while (s1[i] && s2[i])
	{
		if (s1[i] != s2[i])
			return (s1[i] - s2[i]);
		i++;
	}
	return (s1[i] - s2[i]);
}
Mantra
Avance tant qu'égal, return diff des chars courants
Pattern
s1[i] - s2[i] retourne 0 si égal, ±diff sinon. Marche aussi pour gérer les longueurs différentes.
Piège : return 0 au lieu de s1[i] - s2[i] → "abc" == "abcd" à tort.
→ Fiche complète

ft_atoi
Convertir une string en int (avec signe, espaces)
✓ Completed
ft_atoi.c 
int ft_atoi(const char *str)
{
	int i = 0;
	int sign = 1;
	int res = 0;

	while (str[i] == ' ' || (str[i] >= 9 && str[i] <= 13))
		i++;
	if (str[i] == '-' || str[i] == '+')
	{
		if (str[i] == '-')
			sign = -1;
		i++;
	}
	while (str[i] >= '0' && str[i] <= '9')
	{
		res = res * 10 + str[i] - '0';
		i++;
	}
	return (res * sign);
}
Mantra
Whitespace → signe → digits (res*10 + char-'0')
Pattern
3 blocs séquentiels. res * 10 shift à gauche en base 10, puis ajoute le nouveau digit. Applique le signe à la fin.
Piège : '0' et '9' sans quotes → compare aux int 0/9 au lieu des chars '0'/'9'.
→ Fiche complète

Piège : mélanger printf (buffered) et write → \n sort avant les chiffres.

last_word
Afficher le dernier mot d'une string
✓ Completed
last_word.c 
#include <unistd.h>

int	main(int argc, char **argv)
{
	if (argc != 2)
	{
		write(1, "\n", 1);
		return 0;
	}
	int i = 0;
	while (argv[1][i]) i++;
	i--;
	while (i >= 0 && (argv[1][i] == ' ' || argv[1][i] == '\t'))
		i--;
	while (i >= 0 && argv[1][i] != ' ' && argv[1][i] != '\t')
		i--;
	i++;
	while (argv[1][i] != ' ' && argv[1][i] != '\t' && argv[1][i])
	{
		write(1, &argv[1][i], 1);
		i++;
	}
	write(1, "\n", 1);
}
Mantra
Skip trailing spaces → recule sur mot → write forward
Pattern
Parcours INVERSE puis forward. Condition i >= 0 AVANT argv[1][i] dans le && pour éviter UB.
Piège : confusion OR/AND avec != / == (De Morgan). Règle magique : == va avec ||, != va avec &&.
→ Fiche complète

ft_strrev
Inverser une string in-place
✓ Completed
ft_strrev.c 
char	*ft_strrev(char *str)
{
	int i = 0;
	int j = 0;
	char tmp;
	char jayle;

	while (str[i]) i++;
	i--;
	while (i >= j)
	{
		tmp = str[i];
		jayle = str[j];
		str[j] = tmp;
		str[i] = jayle;
		j++;
		i--;
	}
	return (str);
}
Mantra
Deux indices (début, fin), swap avec tmp, converge
Pattern
Two-pointer convergent. Marche pour pair ET impair (quand i == j, sort sans swap).
Piège : swap sans tmp → écrasement. str[i] = str[j] d'abord perd str[i].
→ Fiche complète

alpha_mirror
Miroir alphabétique : a↔z, A↔Z, b↔y...
✓ Completed
alpha_mirror.c 
#include <unistd.h>

int main (int argc, char **argv)
{
	int i = 0;
	int c = 219;   // 'a' + 'z'
	int b = 155;   // 'A' + 'Z'

	if (argc != 2)
	{
		write(1, "\n", 1);
		return 0;
	}
	while (argv[1][i])
	{
		if (argv[1][i] >= 'a' && argv[1][i] <= 'z')
		{
			argv[1][i] = c - argv[1][i];
			write(1, &argv[1][i], 1);
		}
		else if (argv[1][i] >= 'A' && argv[1][i] <= 'Z')
		{
			argv[1][i] = b - argv[1][i];
			write(1, &argv[1][i], 1);
		}
		else
			write(1, &argv[1][i], 1);
		i++;
	}
	write (1, "\n", 1);
}
Mantra
219 - c pour minuscules, 155 - c pour majuscules
Pattern
Reflection autour du milieu. 'a' + 'z' = 219. Donc mirror de c = 219 - c (dans l'intervalle a-z).
Piège : oublier de gérer minuscules ET majuscules séparément → seul un des deux cas marche.
→ Fiche complète

is_power_of_2
Check si un nombre est une puissance de 2
✓ Completed
is_power_of_2.c 
int is_power_of_2(unsigned int n)
{
	if (n == 0)
		return (0);
	while (n > 1)
	{
		if (n % 2 != 0)
			return (0);
		n = n / 2;
	}
	return (1);
}
Mantra
Divise par 2 en boucle. Si reste non-zero → no. Fini à 1 → yes.
Pattern
Alternative bitwise : return n && !(n & (n-1)). Plus rapide mais moins lisible.
Piège : n / 1 au lieu de n / 2 → boucle infinie.
→ Fiche complète

char	*ft_strdup(char *src)
{
	int i = 0;
	while (src[i])
		i++;

	int j = 0;
	while (src[j])

}

Pattern
Premier contact avec malloc. +1 pour le \0. Toujours return char *, pas char.
Piège : oublier le \0 final → la string a des octets random après la fin → UB.

✓ Completed
ft_strcspn.c 
#include <stddef.h>

size_t	ft_strcspn(const char *s, const char *reject)
{
	int i = 0;
	int j = 0;

	while (s[i])
	{
		while (reject[j])
		{
			if (s[i] == reject[j])
				return (i);
			j++;
		}
		j = 0;
		i++;
	}
	return (i);
}
Mantra
Double while : chaque char de s vs reject entier
Pattern
Parent de union / inter. Return immédiat au premier match. Si boucle finit → retourne la longueur totale.
Piège : oublier de reset j = 0 à chaque tour externe → scan incomplet.

int main (int argc, char **argv)
{
	if (argc != 2)
	{
		write(1, "\n", 1);
		return 0;
	}
	int i = 0;
	while (argv[1][i])
	{
		while (argv[1][i] != '_' && argv[1][i])
		{
			write(1, &argv[1][i], 1);
			i++;
		}
		if (argv[1][i] == '_')
		{
			i++;
			argv[1][i] = argv[1][i] - 32;
			write(1, &argv[1][i], 1);
			i++;
		}
	}
	write(1, "\n", 1);
}
Mantra
Underscore → skip + char suivant - 32. Sinon write tel quel.
Pattern
- 32 convertit minuscule en majuscule. Le _ signale "transforme le PROCHAIN char".
Piège : double i++ mal placé → certaines lettres dupliquées ou skippées.

Deux pointeurs : s1 est-il sous-séquence de s2 ?
✓ Completed
wdmatch.c 
#include <unistd.h>

int main(int argc, char **argv)
{
	if (argc != 3)
	{
		write(1, "\n", 1);
		return 0;
	}
	int i = 0;
	int j = 0;
	while (argv[2][i])
	{
		if (argv[1][j] == argv[2][i])
		{
			j++;
		}
		i++;
		if (argv[1][j] == '\0')
		{
			j = 0;
			while (argv[1][j])
			{
				write(1, &argv[1][j], 1);
				j++;
			}
			write(1, "\n", 1);
			return 0;
		}
	}
	write(1, "\n", 1);
	return 0;
}
Mantra
i avance toujours, j seulement quand ça match
Pattern
Two-pointer async. j a un double usage : d'abord curseur de match, puis curseur d'impression (reset à 0).
Piège : confusion argv[1] / argv[2]. Le while externe parcourt la SOURCE (s2), pas la CIBLE (s1).

#include <unistd.h>

void	print_bits(unsigned char octet)
{
	int		i = 7;
	char	c;

	while (i >= 0)
	{
		c = ((octet >> i) & 1) + '0';
		write(1, &c, 1);
		i--;
	}
}
Mantra
shift → & 1 → + '0' → write, 8 fois
Pattern
Shift & mask. (octet >> i) & 1 isole le bit i. + '0' convertit en char affichable.
Piège : while (i--) avec i = 7 rate le bit 7. Démarrer à i = 8 à cause du post-décrément.

reverse_bits
Inverser les 8 bits d'un byte
● Joker
reverse_bits.c 
unsigned char	reverse_bits(unsigned char octet)
{
	unsigned char	res = 0;
	int		i = 0;

	while (i < 8)
	{
		res = res << 1;
		res = res | (octet & 1);
		octet = octet >> 1;
		i++;
	}
	return (res);
}
Mantra
Pop LSB d'octet, push à gauche de res, 8 fois
Pattern
Équivalence arithm ↔ bitwise : *2 ≡ <<1, /2 ≡ >>1, %2 ≡ & 1. Si tu bloques sur les bits, passe par les maths.
Piège : res * 2 APRÈS l'ajout → décale aussi le nouveau bit.
→ Fiche complète

swap_bits
Échanger les 4 bits hauts avec les 4 bits bas
● Joker
swap_bits.c 
#include <unistd.h>

unsigned char	swap_bits(unsigned char octet)
{
	return ((octet >> 4) | (octet << 4));
}
Mantra
>> 4 | << 4 = swap nibbles
Pattern
Shift + OR = fusion de zones disjointes. Le >> 4 descend le nibble haut, << 4 monte le nibble bas. OR fusionne.
Piège : char sans unsigned → sign extension du right shift pollue le résultat.

int	main(int argc, char **argv)
{
	if (argc != 3)
	{
		write(1, "\n", 1);
		return 0;
	}
	int i = 0, j = 0, found = 0;
	while (argv[1][i])
	{
		j = 0; found = 0;
		while (j < i)
		{
			if (argv[1][j] == argv[1][i])
				found = 1;
			j++;
		}
		if (found == 0)
			write(1, &argv[1][i], 1);
		i++;
	}
	i = 0;
	while (argv[2][i])
	{
		j = 0; found = 0;
		while (argv[1][j])
		{
			if (argv[1][j] == argv[2][i])
				found = 1;
			j++;
		}
		j = 0;
		while (j < i)
		{
			if (argv[2][j] == argv[2][i])
				found = 1;
			j++;

		if (found == 0)

}

Phase s1 : look behind. Phase s2 : s1 entier + s2 avant.

inter
Intersection s1 ∩ s2, dans l'ordre de s1, sans doublons
● Joker
inter.c 
#include <unistd.h>

int	main(int argc, char **argv)
{
	if (argc != 3)
	{
		write(1, "\n", 1);
		return 0;
	}
	int i = 0, j = 0, dupe = 0, in_s2 = 0;
	while (argv[1][i])
	{
		j = 0; dupe = 0;
		while (j < i)
		{
			if (argv[1][j] == argv[1][i])
				dupe = 1;
			j++;
		}
		j = 0; in_s2 = 0;
		while (argv[2][j])
		{
			if (argv[2][j] == argv[1][i])
				in_s2 = 1;
			j++;
		}
		if (dupe == 0 && in_s2 == 1)
			write(1, &argv[1][i], 1);
		i++;
	}
	write(1, "\n", 1);
	return 0;
}
Mantra
dupe == 0 && in_s2 == 1 → write
Pattern
Multi-flag. 2 conditions indépendantes combinées en &&. Une seule phase (s1 seul) — l'output sort dans l'ordre de s1.
Piège : || au lieu de && → on écrit des chars soit non-dupliqués soit dans s2, pas les deux.
→ Fiche complète
